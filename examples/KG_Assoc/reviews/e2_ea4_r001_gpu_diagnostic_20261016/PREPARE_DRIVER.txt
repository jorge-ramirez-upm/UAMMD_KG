 for row in e2.campaign.load_json(p)['streams']:
  chemical.add(row['chemistry_seed']);requested.add(row['md_seed']);effective.add(e2.validation.effective_md_seed(row['md_seed']))

for row in reservation['streams']:
 row['chemistry_seed']+=200000;row['md_seed']+=200000
new_chemical=[x['chemistry_seed'] for x in reservation['streams']];new_md=[x['md_seed'] for x in reservation['streams']];new_effective=[e2.validation.effective_md_seed(x) for x in new_md]

e2.exclusive_json(policy,reservation)
