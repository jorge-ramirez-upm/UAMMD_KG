"""Read-only fifteen-record inventory and exception binding audit; no GPU authority."""
import csv
from pathlib import Path
import runpy
h=runpy.run_path(str(Path(__file__).with_name('AUTHENTICATE.txt')))
e2=h['e2'];import kg_assoc_e2_registry as registry
D=h['D']
rows,attempt,plan=h['authenticate']()
r=e2.campaign.load_json(D/'REGISTRATION.log');path=Path(r['acceptance'])
assert e2.campaign.sha256(path)==r['sha256'] and path.read_bytes()==(D/'ACCEPTANCE_RECORD.log').read_bytes()
record=e2.campaign.load_json(path);registry.exclusive_validate(record,[e2.system_by_id(h['SID'])])
assert record['qualification']==(D/'QUALIFICATION.txt').read_text().strip()
assert r['registered_acceptances']==15 and r['scientific_hold_resolved'] is False
assert r['user_approved_scientific_exception'] is True and r['GPU_steps_executed']==0
assert r['registry_selected'] is False and r['registry_activated'] is False
for row,history in zip(rows,record['histories']):
    assert row==r['exact_endpoints'][int(row['replica'])-1]
    assert history['end_step']==88000000 and history['prefix']+'.restart.lammpsdat'==row['restart']
    validation=e2.campaign.load_json(D/f"REVALIDATED_r{int(row['replica']):03d}.log")
    assert validation['exact_endpoint']==row and validation['MD_steps_executed_by_this_check']==0
    assert validation['validation']==e2.campaign.load_json(row['receipt'])['validation']
    assert validation['validation']['oracle_sweeps']==320000
old=list(csv.DictReader((D.parent/'e2_ea8_formal_acceptance_20261023/ACCEPTANCE_INVENTORY.csv').open()))
assert len(old)==14
for row in old:e2.check_hashes({row['record']:row['sha256']})
records=sorted(path.parent.glob('*.json'));assert len(records)==15
systems=set()
for path in records:
    item=e2.campaign.load_json(path);sid,=item['permitted_systems']
    registry.exclusive_validate(item,[e2.system_by_id(sid)]);systems.add(sid)
assert len(systems)==15 and h['SID'] in systems
if (D/'INTEGRITY.sha256').exists():e2.check_hashes(e2.campaign.load_json(D/'INTEGRITY.sha256'))
assert not e2.subprocess.check_output(['git','diff',h['START'],'--','src','examples/KG',
    'examples/KG_Assoc/kg_assoc_production.cu','examples/KG_Assoc/kg_assoc_campaign_production.cu',
    'examples/KG_Assoc/kg_assoc_kinetics.cuh','examples/KG_Assoc/campaign/kg_assoc_e2_validate.py'])
print('PASS exact user-excepted Ea10 88M triplet; original assessments and fourteen acceptances unchanged')
print('PASS fifteen qualified systems; all formal records authenticate; no registry selection/activation')
print('PASS',len(e2.campaign.load_json(D/'PRESERVED.sha256')),'preserved bindings; no GPU/MD or frozen-source changes')
print('ACCEPTANCE',r['acceptance'],r['sha256'])
print('HEAD',e2.campaign.git_sha())
