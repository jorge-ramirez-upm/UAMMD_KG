# One campaign execution only. Never rerun this driver or resume its immutable attempts.
import json,sys
from pathlib import Path
r=Path.cwd();root=r/'examples/KG_Assoc/kg_assoc_100m';w=root/'aggregate/e2/ea_transfer_remaining_smoke_20261017_v1'
sys.path.insert(0,str(r/'examples/KG_Assoc/campaign'))
import kg_assoc_e2 as e2
import kg_assoc_e2_transfer as transfer
prepared=e2.campaign.load_json(w/'PREPARED.json')
expected=[(transfer.REFERENCE,[2,3])]+[(sid,[1,2,3]) for sid in transfer.TARGETS]
assert [(item['system'],item['selected_replicas']) for item in prepared['plans']]==expected
assert prepared['source_git_sha']==e2.campaign.git_sha()
assert sum(len(replicas)*10000 for _,replicas in expected)==prepared['maximum_additional_MD_steps']==110000
e2.clean_tree();e2.check_build(root/'builds/readback_rounding_cpu_20261016_v2')
for item in prepared['plans']:
 attempt=Path(item['attempt']);plan=e2.check_plan(root,attempt,live=True)
 assert e2.campaign.sha256(attempt/'plan.json')==item['plan_sha256']
 assert e2.campaign.sha256(attempt/'approval.json')==item['approval_sha256']
 e2.scoped_launch_approval(attempt/'approval.json',attempt,'e2_derived_smoke',plan,None,item['selected_replicas'])
 assert plan['steps']==10000 and all((entry['start_step'],entry['end_step'])==(8000000,8010000) for entry in e2.launch_scope(plan,None,item['selected_replicas']))
 assert all(not list((attempt/f'r{replica:03d}').iterdir()) for replica in (1,2,3))
completed=[]
try:
 for item in prepared['plans']:
  print('LAUNCH',item['system'],item['selected_replicas'],flush=True)
  attempt=Path(item['attempt'])
  e2.launch(root,attempt,execute=True,approval=attempt/'approval.json',replicas=item['selected_replicas'])
  for replica in item['selected_replicas']:
   receipt=e2.campaign.load_json(attempt/f'r{replica:03d}/complete.json')
   assert receipt['status']=='validated' and receipt['end_step']==8010000
   completed.append(dict(system=item['system'],replica=replica,additional_MD_steps=10000))
  assert sum(entry['additional_MD_steps'] for entry in completed)<=110000
  print('BRANCH_PASS',item['system'],'aggregate_MD_steps',len(completed)*10000,flush=True)
 assert len(completed)==11 and not list((Path(prepared['plans'][0]['attempt'])/'r001').iterdir())
 e2.exclusive_json(w/'CAMPAIGN_RESULT.json',dict(status='passed',completed=completed,additional_MD_steps=110000,finished_at=e2.timestamp(),historical_Ea4_r001_not_executed=True))
 print('ALL_ELEVEN_PASS',110000,flush=True)
except BaseException as error:
 e2.exclusive_json(w/'CAMPAIGN_STOP.json',dict(status='stopped_at_first_failure',error=str(error),system=item['system'],selected_replicas=item['selected_replicas'],completed_branches_histories=completed,finished_at=e2.timestamp(),no_retry=True))
 print('CAMPAIGN_STOP',repr(error),flush=True)
 raise
