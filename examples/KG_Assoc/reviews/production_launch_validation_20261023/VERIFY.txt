"""Read-only CPU publication/preservation audit; no native execution."""
import json
from pathlib import Path
import runpy
import subprocess

D = Path(__file__).resolve().parent
G = runpy.run_path(str(D / 'DRIVER.txt'))
e = G['e2']
c = G['c']
e.check_hashes(c.load_json(D / 'PRESERVED.sha256'))
if (D / 'INTEGRITY.sha256').exists():
    e.check_hashes(c.load_json(D / 'INTEGRITY.sha256'))
authority, rows = G['load_authority'](G['INDEX'])
assert len(rows) == 15 and sum(h['steps'] for h in rows) == 1500000
assert all(G['A']['status'](h, authority) == 'prepared' for h in rows)
assert all({p.name for p in Path(h['output_prefix']).parent.iterdir()} == {'run.json'} for h in rows)
paths = sorted((G['ROOT'] / 'banks/e2_acceptance').glob('*.json'))
assert len(paths) == 15
for path in paths:
    record = c.load_json(path)
    G['A']['registry'].exclusive_validate(record, [e.system_by_id(record['permitted_systems'][0])])
frozen = G['P']['check']()
assert len(frozen['histories']) == 45
assert all({p.name for p in Path(h['output_prefix']).parent.iterdir()} ==
           {'run.json', 'preparation_approval.json'} for h in frozen['histories'])
assert all(p.startswith('examples/KG_Assoc/reviews/production_launch_validation_20261023/') for p in
           subprocess.check_output(['git', 'diff', '--name-only',
             '642de492168b8886cc377a2f11713a715f1cc28a'], text=True).splitlines())
print(json.dumps(dict(status='PASS', qualified_acceptances=15, protected_bindings=
    len(c.load_json(D / 'PRESERVED.sha256')), dormant_production_histories=45,
    bounded_validation_histories=15, maximum_GPU_steps_authorized=1500000,
    GPU_steps_executed_by_assistant=0, native_output_files=0,
    production_execution_approvals=0, production_registry_activated=False,
    identical_pre_post_source_authentication=True,
    source_authorization_sha256=authority['source_authorization_sha256']), indent=2))
