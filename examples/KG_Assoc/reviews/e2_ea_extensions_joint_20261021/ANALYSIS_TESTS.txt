"""CPU-only estimator, conservation, partition and real-data regression tests."""
import json
from pathlib import Path
import runpy
import unittest
import numpy as np

folder = Path(__file__).resolve().parent
old = runpy.run_path(str(folder.parent / 'e2_ea_transfer_24m_joint_20261018/ANALYSIS_DRIVER.txt'))
direct = old['segment'].review.analysis.stationarity.iat
assert direct is not old['fast_iat']
work = Path('examples/KG_Assoc/kg_assoc_100m/aggregate/e2/ea_extensions_joint_20261021_v1/analysis')


class AnalysisTests(unittest.TestCase):
    def test_synthetic_direct_fft(self):
        old['compiler_free_tests']()

    def test_no_windows_cross_restart(self):
        for end in (40000000, 56000000):
            for width in (500000, 1000000, 2000000, 4000000, 8000000):
                boundaries = list(range(24000000, end, width))
                self.assertEqual(boundaries[-1]+width, end)
                self.assertTrue(all(left >= 24000000 for left in boundaries))

    def test_sticker_and_neighbor_conservation(self):
        paths = sorted(work.glob('*_SUMMARY.log'))
        self.assertEqual(len(paths), 6)
        for path in paths:
            h = json.loads(path.read_text())
            exposure = h['exact_integer_step_exposure']
            self.assertEqual(2*exposure['bond_steps']+exposure['free_sticker_steps'],
                             4000*(h['end_step']-h['start_step']))
            self.assertEqual(h['initial']['bonds']+h['counts']['creations']-h['counts']['breaks'],
                             h['final']['bonds'])
            self.assertEqual(h['network_exposure']['neighbor_losses'], h['counts']['neighbor_losses'])

    def test_exact_recovered_receipt(self):
        h = json.loads((work/'EA08_R001_SUMMARY.log').read_text())
        self.assertEqual(h['receipt_kind'], 'independently_validated_recovery')
        self.assertEqual(h['receipt_sha256'],
                         '91a4fa743735a515353d592213d0ee67cf9e2208ca31dc29cfda4b002aaa5091')
        self.assertTrue(Path(h['receipt']).with_name('incomplete.json').exists())


for ea in (8, 10):
    for replica in (1, 2, 3):
        def actual(self, ea=ea, replica=replica):
            rows = json.loads((work/f'EA{ea:02d}_R{replica:03d}_TIMESERIES.log').read_text())[-200:]
            for key in ('intra', 'L1', 'L2', 'mean_degree'):
                values = [row[key] for row in rows]
                times = [row['time'] for row in rows]
                self.assertTrue(np.allclose(direct(values, times), old['fast_iat'](values, times),
                                            rtol=1e-11, atol=1e-11))
        setattr(AnalysisTests, f'test_direct_fft_Ea{ea}_r{replica}', actual)

if __name__ == '__main__':
    unittest.main(verbosity=2)
