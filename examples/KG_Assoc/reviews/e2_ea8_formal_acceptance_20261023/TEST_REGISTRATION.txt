"""CPU-only registrar guards; --registered also audits the immutable final inventory."""
import argparse
import copy
from pathlib import Path
import runpy
import unittest
from unittest import mock

folder = Path(__file__).resolve().parent
helper = runpy.run_path(str(folder/'AUTHENTICATE.txt'))
e2 = helper['e2']
import kg_assoc_e2_registry as registry
registered = False


class RegistrationTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        import csv
        cls.rows = [r for r in csv.DictReader((helper['REVIEW']/'ENDPOINTS.csv').open())
                    if r['system'] == helper['SID']]
        cls.attempt = Path(cls.rows[0]['restart']).parents[1]
        cls.source = cls.attempt/'plan.json'
        cls.plan = e2.campaign.load_json(cls.source)
        cls.system = e2.system_by_id(helper['SID'])
        cls.record = dict(schema_version=1,stage='E2_acceptance',bank_id='F04_N010_RP080_EE08',
            provenance_kind='E2_G1_segment',status='accepted_with_qualification',reviewer='CPU test',
            reviewed_at='CPU test',evidence=str(helper['REVIEW']/'ASSESSMENT.md'),
            evidence_sha256=e2.campaign.sha256(helper['REVIEW']/'ASSESSMENT.md'),
            source_plan=str(cls.source),source_plan_sha256=e2.campaign.sha256(cls.source),
            permitted_use='production_initial_state',permitted_systems=[helper['SID']],
            qualification='qualified initialization; equilibrium unproven',
            numerical_qualification='coupled integrator invariance unproven',publisher_git_sha=e2.campaign.git_sha(),
            histories=[dict(replica=h['replica'],prefix=h['output_prefix'],end_step=h['end_step'],
                chemistry_seed=h['chemistry_seed'],physics=h['physics'],lineage=h['lineage'],
                sha256=e2.hash_files([Path(h['output_prefix']+s) for s in e2.validation.OUTPUTS]+
                                    [Path(h['output_prefix']).parent/'complete.json',cls.source]))
                       for h in cls.plan['histories']])

    def test_exact_reviewed_candidate_uses_established_schema(self):
        registry.exclusive_validate(self.record,[self.system])

    def test_unknown_legacy_extension_ea10_and_wrong_ranges_reject(self):
        original = e2.campaign.load_json
        changes = [{'lineage_mode':'unknown'}, {'lineage_mode':'Ea_24M_extension_v1'},
                   {'steps':8000001}, {'reference_control':True}]
        wrong = copy.deepcopy(self.plan['histories']);wrong[0]['end_step'] = 48000001
        changes.append({'histories':wrong})
        changes.append({'system':dict(self.plan['system'],system_id='F04_N010_RP080_EE08_EA10')})
        for change in changes:
            source = dict(self.plan,**change)
            def load(path):
                return source if str(path) == str(self.source) else original(path)
            with self.subTest(change=list(change)),mock.patch.object(e2.campaign,'load_json',side_effect=load), \
                    self.assertRaises(ValueError):
                registry.exclusive_validate(self.record,[self.system])

    def test_matching_ea10_reviewed_extension_still_cannot_be_accepted(self):
        import csv
        row = next(r for r in csv.DictReader((helper['REVIEW']/'ENDPOINTS.csv').open())
                   if r['system'] == 'F04_N010_RP080_EE08_EA10')
        source_path = Path(row['restart']).parents[1]/'plan.json'
        source = e2.campaign.load_json(source_path)
        bad = copy.deepcopy(self.record)
        bad.update(source_plan=str(source_path),source_plan_sha256=e2.campaign.sha256(source_path),
                   permitted_systems=[source['system']['system_id']])
        bad['histories'] = [dict(replica=h['replica'],prefix=h['output_prefix'],end_step=h['end_step'],
            chemistry_seed=h['chemistry_seed'],physics=h['physics'],lineage=h['lineage'],
            sha256=e2.hash_files([Path(h['output_prefix']+suffix) for suffix in e2.validation.OUTPUTS]+
                                [Path(h['output_prefix']).parent/'complete.json',source_path]))
                           for h in source['histories']]
        with self.assertRaisesRegex(ValueError,'only the qualified Ea8 48M'):
            registry.exclusive_validate(bad,[e2.system_by_id(source['system']['system_id'])])

    def test_wrong_scope_unqualified_and_endpoint_substitution_reject(self):
        for key,value in [('permitted_use','production'),('permitted_systems',['F04_N010_RP080_EE08_EA10']),
                          ('qualification','')]:
            bad = copy.deepcopy(self.record);bad[key] = value
            with self.assertRaises(ValueError):registry.exclusive_validate(bad,[self.system])
        for key,value in [('prefix',self.record['histories'][1]['prefix']),('end_step',40000000),
                          ('chemistry_seed',1),('lineage',{})]:
            bad = copy.deepcopy(self.record);bad['histories'][0][key] = value
            with self.assertRaises(ValueError):registry.exclusive_validate(bad,[self.system])

    def test_missing_or_corrupt_bound_evidence_reject(self):
        for path in (self.rows[0]['restart'],self.rows[0]['sidecar'],self.rows[0]['receipt']):
            bad = copy.deepcopy(self.record);bad['histories'][0]['sha256'][path] = 'corrupt'
            with self.assertRaises(ValueError):registry.exclusive_validate(bad,[self.system])
        bad = copy.deepcopy(self.record);del bad['histories'][0]['sha256'][self.rows[0]['receipt']]
        with self.assertRaises(ValueError):registry.exclusive_validate(bad,[self.system])

    def test_prior_thirteen_immutable_and_ea10_absent(self):
        prior = e2.campaign.load_json(folder/'PRIOR_ACCEPTANCES.log')
        self.assertEqual(len(prior),13)
        for row in prior:e2.check_hashes({row['path']:row['sha256']})
        files = list((helper['ROOT']/'banks/e2_acceptance').glob('*.json'))
        self.assertEqual(len(files),14 if registered else 13)
        systems = [sid for p in files for sid in e2.campaign.load_json(p)['permitted_systems']]
        self.assertEqual(len(set(systems)),len(files))
        self.assertNotIn('F04_N010_RP080_EE08_EA10',systems)

    def test_administrative_classification_keeps_numerical_and_physics_sources(self):
        registrar = str(e2.HERE/'kg_assoc_e2_registry.py')
        validator = str(e2.HERE/'kg_assoc_e2_validate.py')
        physics = str(e2.HERE.parent/'kg_assoc_kinetics.cuh')
        other = str(e2.HERE.parent/'analysis/kg_assoc_e2_registry.py')
        values = {registrar:'administrative',validator:'numerical',physics:'physics',other:'other'}
        self.assertEqual(e2.scientific_source_hashes(values),
                         {validator:'numerical',physics:'physics',other:'other'})
        self.assertNotEqual(e2.scientific_source_hashes(dict(values,**{validator:'tampered'})),
                            e2.scientific_source_hashes(values))
        self.assertIn(registrar,e2.sources())

    def test_cpu_registration_permission_cannot_authorize_gpu_source_evolution(self):
        with mock.patch.object(e2,'clean_tree'), self.assertRaisesRegex(ValueError,'sources/history changed'):
            e2.live_code(self.plan,{})

    def test_historical_recovery_and_ordinary_child_remain_distinct(self):
        h = self.plan['histories'][0]
        recovery = Path(h['lineage']['parent_receipt'])
        self.assertEqual(e2.campaign.sha256(recovery),
                         '91a4fa743735a515353d592213d0ee67cf9e2208ca31dc29cfda4b002aaa5091')
        self.assertEqual(e2.campaign.sha256(recovery.with_name('incomplete.json')),
                         '14772d19744e253097b4376195210f9591eb6c85df0a012b0ad25f60c46b017c')
        self.assertEqual(e2.campaign.load_json(self.rows[0]['receipt'])['status'],'validated')

    @unittest.skipUnless('--registered' in __import__('sys').argv,'post-publication only')
    def test_published_record_bytes_scope_and_no_replacement(self):
        registration = e2.campaign.load_json(folder/'REGISTRATION.log')
        path = Path(registration['acceptance'])
        self.assertEqual(path.read_bytes(),(folder/'ACCEPTANCE_RECORD.log').read_bytes())
        self.assertEqual(e2.campaign.sha256(path),registration['sha256'])
        record = e2.campaign.load_json(path)
        registry.exclusive_validate(record,[self.system])
        self.assertEqual(record['permitted_systems'],[helper['SID']])
        self.assertEqual(record['permitted_use'],'production_initial_state')
        for phrase in ('finite-step','historical','effective samples','initialization-bias','Ea10','40M'):
            self.assertIn(phrase,record['qualification'])
        with self.assertRaisesRegex(ValueError,'new acceptance filename'):
            registry.accept(helper['ROOT'],path,None,'reviewer',folder/'ACCEPTANCE_RECORD.log',
                            [helper['SID']],'qualified')


if __name__ == '__main__':
    p = argparse.ArgumentParser();p.add_argument('--registered',action='store_true');a = p.parse_args()
    registered = a.registered
    result = unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(RegistrationTests))
    raise SystemExit(not result.wasSuccessful())
