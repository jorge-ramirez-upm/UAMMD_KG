"""Register only the user-approved exact Ea6 24M triplet, using the E2 registrar."""
import csv
from pathlib import Path
import sys
c=Path('examples/KG_Assoc').resolve();d=Path(__file__).resolve().parent;root=c/'kg_assoc_100m'
sys.path.insert(0,str(c/'campaign'))
import kg_assoc_e2 as e2
import kg_assoc_e2_registry as registry
review=c/'reviews/e2_ea_transfer_24m_joint_20261018'
e2.check_hashes(e2.campaign.load_json(review/'INTEGRITY.sha256'))
rows=[r for r in csv.DictReader((review/'ENDPOINTS.csv').open()) if r['Ea']=='6']
assert [int(r['replica']) for r in rows]==[1,2,3]
attempt=Path(rows[0]['restart']).parents[1]
for row in rows:
 assert Path(row['restart']).parents[1]==attempt and row['absolute_step']=='24000000'
 e2.check_hashes({row['restart']:row['restart_sha256'],row['sidecar']:row['sidecar_sha256']})
 revalidated=e2.campaign.load_json(d/f"REVALIDATED_r{int(row['replica']):03d}.log")
 assert revalidated['exact_endpoint']==row and revalidated['validation']['numerical_status']=='valid'
 assert revalidated['MD_steps_executed_by_this_check']==0
sid='F04_N010_RP080_EE08_EA06'
qualification=('Explicit user-approved qualified production initialization of only these three exact 24M Ea6 endpoints. '
 'Inherited corrected-C1 Ea4 preparation and historical provenance limitations remain: historical executable bytes '
 'are unavailable and legacy source/build provenance is less comprehensive than E2-G1. '
 'Equilibrium invariance of the finite-step coupled MD/chemistry algorithm is unproven. '
 'Some slow structural observables have limited effective samples. Replica-specific late excursions remain, '
 'especially r002 intra-star associations. Complete structural decorrelation and initialization-bias removal '
 'have not been demonstrated. This permits production_initial_state only; no registry selection, activation '
 'or production launch. Ea8 and Ea10 remain on scientific hold.')
output=root/'banks/e2_acceptance/F04_N010_RP080_EE08_EA06_24M_qualified_20261018_v1.json'
assert not output.exists(), 'immutable acceptance already exists: authenticate it; never overwrite'
record=registry.accept(root,output,attempt,
 'User explicit Ea6 qualified-acceptance approval, 2026-10-18; registered by Codex',
 review/'ASSESSMENT.md',[sid],qualification)
assert record['permitted_systems']==[sid] and record['permitted_use']=='production_initial_state'
for row,history in zip(rows,record['histories']):
 assert history['prefix']+'.restart.lammpsdat'==row['restart'] and history['end_step']==24000000
 assert history['sha256'][row['restart']]==row['restart_sha256']
 assert history['sha256'][row['sidecar']]==row['sidecar_sha256']
with (d/'ACCEPTANCE_RECORD.log').open('xb') as target:target.write(output.read_bytes())
e2.exclusive_json(d/'REGISTRATION.log',dict(acceptance=str(output),sha256=e2.campaign.sha256(output),
 exact_endpoints=rows,prior_acceptances=12,registered_acceptances=13,registry_selected=False,
 Ea8_hold_retained=True,Ea10_hold_retained=True,GPU_steps_executed=0))
print('REGISTERED',output,e2.campaign.sha256(output))
