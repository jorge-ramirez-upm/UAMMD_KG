import argparse,csv,json

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

parser=argparse.ArgumentParser(description=__doc__)

parser.add_argument('--output',type=Path,required=True)
a=parser.parse_args();a.output.mkdir(exist_ok=False)
result=json.loads((a.analysis/'RESULTS.log').read_text())
assert result['manually_completed_MD_steps']==144000000 and len(result['histories'])==9
colors=['#0072B2','#D55E00','#009E73'];eas=[6,8,10]
records=[]
for h in result['histories']:
 label=f"EA{int(h['Ea']):02d}_R{h['replica']:03d}"
 records.append(dict(history=h,rows=json.loads((a.analysis/(label+'_TIMESERIES.log')).read_text()),
   windows=json.loads((a.analysis/(label+'_WINDOWS.log')).read_text())['windows'],
   renewal=json.loads((a.analysis/(label+'_RENEWAL.log')).read_text())))

def save(figure,name):
 figure.tight_layout()
 figure.savefig(a.output/(name+'.png'),dpi=150)
 figure.savefig(a.output/(name+'.pdf'))
 plt.close(figure)

def grid_metrics(names,filename,windows=False):
 figure,axes=plt.subplots(len(names),3,figsize=(15,2.5*len(names)),squeeze=False)
 for column,ea in enumerate(eas):
  for record in records:
   h=record['history']
   if h['Ea']!=ea:continue
   for axis,name in zip(axes[:,column],names):
    if windows:
     samples=record['windows']
     x=[(s['start_step_exclusive']+s['end_step_inclusive'])/2e6 for s in samples]
     y=[s['metrics'][name]['mean'] for s in samples]
     error=[s['metrics'][name]['sem_estimate'] for s in samples]
     unresolved=[s['metrics'][name]['resolution_warning'] for s in samples]
     axis.errorbar(x,y,yerr=error,color=colors[h['replica']-1],label=f"parent r{h['replica']:03d}",lw=.9,marker='.',ms=3)
     axis.scatter([t for t,bad in zip(x,unresolved) if bad],[v for v,bad in zip(y,unresolved) if bad],marker='x',s=15,color=colors[h['replica']-1])
    else:
     rows=record['rows'];axis.plot([row['step']/1e6 for row in rows],[row[name] for row in rows],lw=.6,alpha=.8,color=colors[h['replica']-1],label=f"parent r{h['replica']:03d}")
    axis.set_ylabel(name);axis.grid(alpha=.2)
  axes[0,column].set_title(f'Ea={ea}; three paired original parent families')
  axes[0,column].legend(fontsize=7)
  axes[-1,column].set_xlabel('Absolute MD step / 1M; branch begins at 8M')
 figure.suptitle('500k windows; IAT SEM; × = unresolved correlation/mixing' if windows else 'Complete derived segment; no correlations stitched across 8M',y=1.001)
 save(figure,filename)
