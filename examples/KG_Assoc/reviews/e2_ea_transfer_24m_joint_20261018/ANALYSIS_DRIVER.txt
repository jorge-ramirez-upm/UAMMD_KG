def partition(neighbors):
    unseen = set(neighbors)
    components = []
    while unseen:
        first = min(unseen)
        unseen.remove(first)
        frontier = [first]
        component = [first]
        while frontier:
            star = frontier.pop()
            for neighbor in neighbors[star]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    frontier.append(neighbor)
                    component.append(neighbor)
        components.append(sorted(component))
    components.sort(key=lambda members: (-len(members), members[0]))
    categories = {star: 'giant' for star in components[0]}
    for members in components[1:]:
        categories.update({star: 'isolated' if len(members) == 1 else 'satellite' for star in members})
    # Isolation remains degree zero even if a singleton wins the largest-component tie.
    categories.update({star: 'isolated' for star in neighbors if not neighbors[star]})
    return components, categories


def network_renewal(initial_pairs, raw, system, start, end, sample_steps):
    stars = sorted(system['stars'])
    mapping = system['atom_to_star']
    def edge(pair):
        return tuple(sorted(mapping[atom] for atom in pair))
    original_pairs = set(initial_pairs)
    active = set(initial_pairs)
    surviving_pairs = set(initial_pairs)
    multiplicity = Counter(edge(pair) for pair in active if mapping[pair[0]] != mapping[pair[1]])
    original_edges = set(multiplicity)
    surviving_edges = set(multiplicity)
    neighbors = {star: set() for star in stars}
    for a, b in multiplicity:
        neighbors[a].add(b)
        neighbors[b].add(a)
    _, categories = partition(neighbors)
    starts = {star: start for star in stars}
    initial_categories = categories.copy()
    episodes = []
    gains = Counter()
    losses = Counter()
    gain_sets = defaultdict(set)
    loss_sets = defaultdict(set)
    reactions = Counter()
    sampled = []
    index = 0
    def close(star, at, right=False):
        episodes.append(dict(star=star, category=categories[star], start_step=starts[star],
            end_step=at, duration_steps=at-starts[star],
            left_censored=starts[star] == start, right_censored=right))
    for boundary in sample_steps:
        assert start <= boundary <= end
        while index < len(raw) and raw[index][0] <= boundary:
            step, kind, pair = raw[index]
            assert start < step <= end and kind in ('C', 'B')
            a, b = edge(pair)
            for star in set((a, b)):
                reactions[star] += 1
            if kind == 'C':
                assert pair not in active
                active.add(pair)
            else:
                assert pair in active
                active.remove(pair)
                surviving_pairs.discard(pair)
            topology_changed = False
            if a != b:
                if kind == 'C':
                    if multiplicity[(a,b)] == 0:
                        gains[(a,b)] += 1
                        gain_sets[a].add(b)
                        gain_sets[b].add(a)
                        neighbors[a].add(b)
                        neighbors[b].add(a)
                        topology_changed = True
                    multiplicity[(a,b)] += 1
                else:
                    multiplicity[(a,b)] -= 1
                    assert multiplicity[(a,b)] >= 0
                    if multiplicity[(a,b)] == 0:
                        del multiplicity[(a,b)]
                        losses[(a,b)] += 1
                        loss_sets[a].add(b)
                        loss_sets[b].add(a)
                        surviving_edges.discard((a,b))
                        neighbors[a].remove(b)
                        neighbors[b].remove(a)
                        topology_changed = True
                if topology_changed:
                    _, updated = partition(neighbors)
                    for star in stars:
                        if updated[star] != categories[star]:
                            close(star, step)
                            starts[star] = step
                    categories = updated
            index += 1
        components, _ = partition(neighbors)
        sampled.append(dict(step=boundary, surviving_initial_bonds=len(surviving_pairs),
            surviving_initial_neighbor_edges=len(surviving_edges),
            current_initial_bond_overlap=len(original_pairs & active),
            current_initial_neighbor_overlap=len(original_edges & set(multiplicity)),
            reacted_stars=len(reactions), neighbor_gain_stars=len(gain_sets),
            neighbor_loss_stars=len(loss_sets),
            isolated_ids=sorted(star for star in stars if not neighbors[star]),
            satellite_components=[members for members in components[1:] if len(members)>1]))
    assert index == len(raw)
    for star in stars:
        close(star, end, right=True)
    per_star = []
    grouped = defaultdict(list)
    for episode in episodes:
        grouped[episode['star']].append(episode)
    for star in stars:
        values = grouped[star]
        assert sum(item['duration_steps'] for item in values) == end-start
        residence = {category: sum(item['duration_steps'] for item in values if item['category']==category)
                     for category in ('giant','satellite','isolated')}
        visits = {category: sum(item['category']==category for item in values)
                  for category in residence}
        nonzero_gap_returns = {}
        for category in residence:
            visits_of_category = [item for item in values if item['category'] == category]
            nonzero_gap_returns[category] = sum(current['start_step'] > previous['end_step']
                for previous, current in zip(visits_of_category, visits_of_category[1:]))
        per_star.append(dict(star=star, initial_category=initial_categories[star],
            final_category=categories[star], accepted_reactions=reactions[star],
            gained_neighbor_ids=sorted(gain_sets.get(star, set())),
            lost_neighbor_ids=sorted(loss_sets.get(star, set())), residence_steps=residence,
            visits=visits, returns={key:max(0,value-1) for key,value in visits.items()},
            nonzero_gap_returns=nonzero_gap_returns,
            positive_duration_visits={key:sum(item['category']==key and item['duration_steps']>0
                                            for item in values) for key in residence}))
    return dict(sampled=sampled, episodes=episodes, stars=per_star,
        distinct_edge_gains=sum(gains.values()), distinct_edge_losses=sum(losses.values()),
        unique_gained_edges=len(gains), unique_lost_edges=len(losses),
        initial_edges=len(original_edges), initial_bonds=len(original_pairs),
        surviving_initial_bonds=len(surviving_pairs), surviving_initial_edges=len(surviving_edges),
        boundary_policy='Initial episodes and bonds left-censored; endpoint episodes right-censored; same-step accepted-event order retained with zero duration.',
        giant_tie_policy='Largest component, then smallest member ID; isolated always means degree zero.')


def self_test():
    system=dict(stars=[1,2,3,4],atom_to_star={i:(i-1)//2+1 for i in range(1,9)})
    raw=[(100,'C',(4,5)),(200,'B',(1,3)),(300,'C',(1,7)),(400,'B',(4,5))]
    result=network_renewal([(1,3)],raw,system,0,500,[0,100,200,300,400,500])
    assert (result['distinct_edge_gains'],result['distinct_edge_losses'])==(2,2)
    assert result['surviving_initial_bonds']==result['surviving_initial_edges']==0
    assert [row['residence_steps']['isolated'] for row in result['stars']]==[100,100,200,300]
    assert result['stars'][2]['returns']['isolated']==1
    assert result['stars'][1]['residence_steps']['satellite']==100
    assert integer_exposure(raw,1,0,500,8)==dict(bond_steps=700,free_sticker_steps=2600,final_bonds=1)
    # A same-step excursion contributes zero residence but remains an ordered event.
    zero=network_renewal([],[(100,'C',(1,3)),(100,'B',(1,3))],system,0,200,[0,100,200])
    assert all(star['residence_steps']['isolated']==200 for star in zero['stars'])
    assert zero['distinct_edge_gains']==zero['distinct_edge_losses']==1
    assert all(star['nonzero_gap_returns']['isolated']==0 for star in zero['stars'])


def statistics_window(rows, events, start, end, metrics):
    result=segment.review.window(rows,events,start,end,metrics=metrics)
    for category in ('metrics','activity'):
        for item in result[category].values():
            constant=len(set(item['blocks']))==1 and item['sem_estimate']==0
            item['constant_not_mixing_evidence']=constant
            if constant:
                item['effective_samples_estimate']=None
                item['tau_time']=None
            item['resolution_warning'] |= constant or (item['effective_samples_estimate'] or 0)<10
    return result

def merge(output):
    histories=[];table=[]
    state=e2.campaign.load_json(WORK/'state.json')
    for job in state['jobs']:
        label=f"EA{int(job['system'][-2:]):02d}_R{job['replica']:03d}"

        histories.append(record['history']);table.extend(record['table'])

    old=e2.campaign.load_json(ASSOC/'reviews/e2_ea_transfer_smoke_complete_20261017/RESULTS.log')['histories']
    pairings=[];contrasts=[]
    for replica in (1,2,3):
        family=[h for h in histories if h['replica']==replica]
        previous=[h for h in old if h['replica']==replica]
        assert len({h['initial_device_payload_sha256'] for h in family+previous})==1
        assert len({h['parent']['sha256'][h['parent']['snapshot']] for h in family})==1
        pairings.append(dict(parent_family=replica,barriers=[6,8,10],initial_device_payload_sha256=family[0]['initial_device_payload_sha256'],identical_to_all_smoke_barriers=True))
        control=next(h for h in previous if h['Ea']==4)
        for h in family:
            h['conditional_reference_prediction']=dict(q_ratio=h['q']/control['q'],
                expected_creations=control['counts']['creations']/control['free_sticker_time']*h['free_sticker_time']*h['q']/control['q'],
                expected_breaks=control['counts']['breaks']/control['active_bond_time']*h['bond_time']*h['q']/control['q'],
                control_attempt=control['attempt'],limit='Descriptive short-control conditional rate extrapolation; changing eligibility and network geometry are not normalized away.')
    kinetic_pairings = []
    for replica in (1,2,3):
        low = next(h for h in histories if h['Ea']==6 and h['replica']==replica)
        for ea in (8,10):
            high = next(h for h in histories if h['Ea']==ea and h['replica']==replica)
            ratio = high['q']/low['q']
            kinetic_pairings.append(dict(parent_family=replica,Ea=ea,reference_Ea=6,q_ratio=ratio,
                creation_rate_ratio=(high['counts']['creations']/high['free_sticker_time']) /
                                    (low['counts']['creations']/low['free_sticker_time']),
                rupture_rate_ratio=(high['counts']['breaks']/high['bond_time']) /
                                   (low['counts']['breaks']/low['bond_time']),
                predicted_creations=low['counts']['creations']/low['free_sticker_time']*
                                    high['free_sticker_time']*ratio,
                predicted_breaks=low['counts']['breaks']/low['bond_time']*high['bond_time']*ratio,
                observed_creations=high['counts']['creations'],observed_breaks=high['counts']['breaks'],
                qualification='Exposure normalization omits geometric eligibility and competing acceptance.'))
    for ea in (8,10):
        for name in histories[0]['phases']['last_quarter']['metrics']:
            differences=[]
            for replica in (1,2,3):
                high=next(h for h in histories if h['Ea']==ea and h['replica']==replica)
                low=next(h for h in histories if h['Ea']==6 and h['replica']==replica)
                differences.append(high['phases']['last_quarter']['metrics'][name]['mean']-low['phases']['last_quarter']['metrics'][name]['mean'])
            contrasts.append(dict(Ea_minus_Ea6=ea,observable=name,paired_differences=differences,
                **segment.review.analysis.dynamics.replica_statistics(differences)))
    with (output/'WINDOWS.csv').open('x',newline='') as stream:
        writer=csv.DictWriter(stream,fieldnames=list(table[0]));writer.writeheader();writer.writerows(table)
    e2.exclusive_json(output/'RESULTS.log',dict(starting_sha='deb0ef3797917ee1cd890d19765ec654ecc3efce',launch_source_sha=histories[0]['source_git_sha'],
        manually_completed_GPU_histories=9,manually_completed_MD_steps=144000000,assessment_GPU_steps=0,device_readbacks=9,candidate_comparisons=1440000,
        statistical_preparation_units=3,pairings=pairings,histories=histories,paired_kinetic_comparisons=kinetic_pairings,paired_last_quarter_contrasts=contrasts,
        scientific_status='review_required_no_automatic_acceptance',formal_acceptance_registered=False))
    print('ALL_NINE_REPLAYED',flush=True)



def fast_iat(values, times):
    """Same biased ACF and first-nonpositive truncation; FFT reduces CPU work only."""
    intervals = np.diff(times)
    if not len(intervals) or max(intervals)-min(intervals)>1e-10*max(intervals):
        return float('nan'), float('nan')
    centered = np.asarray(values, dtype=float)-statistics.mean(values)
    variance = float(np.dot(centered, centered))
    if variance == 0:
        return .5, len(values)
    size = 1 << (2*len(values)-1).bit_length()
    spectrum = np.fft.rfft(centered, n=size)
    correlations = np.fft.irfft(spectrum*spectrum.conjugate(), n=size)[:len(values)]/variance
    nonpositive = np.flatnonzero(correlations[1:] <= 0)
    stop = int(nonpositive[0])+1 if len(nonpositive) else len(values)
    tau = .5+float(sum(correlations[1:stop]))
    return tau, len(values)/(2*tau)


def compiler_free_tests():
    self_test()
    original=segment.review.analysis.stationarity.iat
    rng=np.random.default_rng(20261018)
    fixtures=[np.zeros(80), np.arange(120), np.tile([-1.,1.],60),
              rng.normal(size=150), np.cumsum(rng.normal(size=180))]
    for values in fixtures:
        times=list(np.arange(len(values))*10.)
        expected=original(list(values),times)
        observed=fast_iat(list(values),times)
        assert np.allclose(expected,observed,rtol=1e-12,atol=1e-12),(expected,observed)
    exposure_helpers['self_test']()
    print('PASS 5 FFT/direct IAT fixtures; exact event residence/returns; distinct-edge exposure')
