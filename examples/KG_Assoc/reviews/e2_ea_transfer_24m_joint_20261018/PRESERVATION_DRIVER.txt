"""Read-only preservation snapshot/check, including manual raw outputs; no MD."""
import argparse
import json
from pathlib import Path
import subprocess
import sys
r=Path.cwd();c=r/'examples/KG_Assoc';d=Path(__file__).resolve().parent
sys.path.insert(0,str(c/'campaign'))
import kg_assoc_e2 as e2
p=argparse.ArgumentParser();p.add_argument('--snapshot',action='store_true');a=p.parse_args()
if a.snapshot:
    paths={r/name for name in subprocess.check_output(['git','ls-files'],text=True).splitlines()}
    old=c/'kg_assoc_100m/aggregate/e2/ea_transfer_16m_20261017_v1/BASELINE.sha256'
    paths.update(Path(name) for name in e2.campaign.load_json(old))
    state=c/'kg_assoc_100m/aggregate/e2/manual_ea16m_v1/state.json'
    paths.add(state)
    for job in e2.campaign.load_json(state)['jobs']:
        paths.update(path for path in Path(job['attempt']).rglob('*') if path.is_file())
    paths.update(path for path in (c/'kg_assoc_100m/builds').rglob('*') if path.is_file())
    e2.exclusive_json(d/'PRESERVED.sha256',e2.hash_files(sorted(path for path in paths if path.is_file())))
    print('SNAPSHOT',len(paths))
else:
    bindings=e2.campaign.load_json(d/'PRESERVED.sha256');e2.check_hashes(bindings)
    records=e2.campaign.load_json(c/'reviews/e2_ea_transfer_gpu_diagnostic_retest_20261016/PREFLIGHT.json')['acceptance_records']
    assert len(records)==12
    assert set((c/'kg_assoc_100m/banks/e2_acceptance').glob('*.json'))=={r/item['record'] for item in records}
    for item in records:
        path=r/item['record'];assert e2.campaign.sha256(path)==item['sha256']
        record=e2.campaign.load_json(path)
        e2.check_hashes({record['evidence']:record['evidence_sha256'],record['source_plan']:record['source_plan_sha256']})
        for entry in record['histories']:e2.check_hashes(entry['sha256'])
    e2.check_build(c/'kg_assoc_100m/builds/ea_transfer_16m_20261017_v1')
    print(json.dumps(dict(preserved_bindings=len(bindings),formal_acceptances_and_payloads=12,
        native_build_authenticated=True,GPU_jobs_launched=0,protected_directories_unchanged=True),indent=2))
