"""Deterministic review-only tests. No native executable or GPU invocation."""
import os
import json
import math
from pathlib import Path
import runpy
import unittest
import numpy as np
D=Path(__file__).resolve().parent
F=runpy.run_path(str(D/'ANALYSIS_DRIVER.txt'))
W=Path(os.environ.get('EA24M_ANALYSIS',str(F['ROOT']/'aggregate/e2/ea_transfer_24m_joint_20261018_v1/analysis')))
class AnalysisTests(unittest.TestCase):
    def test_direct_and_fft_synthetic(self):F['compiler_free_tests']()
    def test_iat_shift_and_scale(self):
        values=list(np.sin(np.arange(300)*.07));times=list(np.arange(300)*10.)
        self.assertTrue(np.allclose(F['fast_iat'](values,times),F['fast_iat']([3*x+9 for x in values],times),rtol=1e-12))
    def test_nonuniform_cadence_is_unresolved(self):
        self.assertTrue(all(math.isnan(v) for v in F['fast_iat']([1,2,3],[0,1,3])))
    def test_long_trend_does_not_report_independent_samples(self):
        tau,effective=F['fast_iat'](list(range(200)),list(range(200)))
        self.assertLess(effective,10);self.assertGreater(tau,20)
    def test_all_exact_exposures_conserve_stickers(self):
        for p in W.glob('*_RECORD.log'):
            h=json.loads(p.read_text())['history'];x=h['exact_integer_step_exposure']
            self.assertEqual(2*x['bond_steps']+x['free_sticker_steps'],4000*16000000)
            self.assertEqual(x['final_bonds'],h['final']['bonds'])
for ea in (6,8,10):
    for replica in (1,2,3):
        def test_real(self,ea=ea,replica=replica):
            rows=json.loads((W/f'EA{ea:02d}_R{replica:03d}_TIMESERIES.log').read_text())[-200:]
            for key in ('intra','L1','mean_degree'):
                values=[r[key] for r in rows];times=[r['time'] for r in rows]
                self.assertTrue(np.allclose(F['fast_iat'](values,times),F['segment'].review.analysis.stationarity.iat(values,times),rtol=1e-11,atol=1e-11))
        setattr(AnalysisTests,f'test_direct_fft_real_Ea{ea}_r{replica}',test_real)
if __name__=='__main__':unittest.main(verbosity=2)
