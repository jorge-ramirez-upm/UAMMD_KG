class Tests(unittest.TestCase):
    def test_deviatoric_projection_normalization(self):
        rng=np.random.default_rng(20261023)
        for i in range(100):
            x=rng.normal(size=(3,3));x=(x+x.T)/2
            y=rng.normal(size=(3,3));y=(y+y.T)/2
            n=lambda z:np.array([z[0,0]-z[1,1],z[0,0]-z[2,2],z[1,1]-z[2,2]])
            channels=sum(x[i,j]*y[i,j] for i,j in ((0,1),(0,2),(1,2)))/5+np.dot(n(x),n(y))/30
            dx=x-np.eye(3)*np.trace(x)/3;dy=y-np.eye(3)*np.trace(y)/3
            self.assertAlmostEqual(channels,np.sum(dx*dy)/10,places=13)
    def test_normal_difference_dependence(self):
        for xx,yy,zz in ((1.,2.,4.),(-3.,0.,9.),(.2,.7,-.8)):
            self.assertAlmostEqual(xx-zz,(xx-yy)+(yy-zz))
    def test_isotropic_covariance_reproduces_one_shear_scale(self):
        c=2.3;volume=7.2;temperature=1.4
        g=volume*(3*c/5+3*4*c/30)/temperature
        self.assertAlmostEqual(g,volume*c/temperature)
    def test_actual_five_pilot_modulus_and_support_match(self):
        paths=[*Path('p42_rheology_pilot/run.MBA5fA').glob('*.stress_correlator'),
               *Path('p42_rheology_pilot/run.RqADk7').glob('*.stress_correlator')]
        replicas=[]
        for path in paths:
            meta,rows=pilot.read_stress_correlator(str(path));replicas.append(dict(path=str(path),metadata=meta,rows=rows))
        pilot.require_matching_lags(replicas);self.assertEqual(len(replicas),5)
    def test_corrupt_modulus_and_sampling_rejected(self):
        with tempfile.TemporaryDirectory() as temp:
            prefix=Path(temp)/'sample';prefix.with_suffix('.state').write_text('# T=1\n')
            path=prefix.with_suffix('.stress_correlator')
            for header,values in [('# stress_samples=1 step0_sampled=no stress_interval_steps=1 dt=0.01','0 1 1 1 4 4 4 99 1'),
                                  ('# stress_samples=1 step0_sampled=yes stress_interval_steps=1 dt=0.01','0 1 1 1 4 4 4 1 1')]:
                path.write_text(header+'\n'+pilot.COUNT_HEADER+'\n'+values+'\n')
                with self.assertRaises(ValueError):pilot.read_stress_correlator(str(path))
    def test_raw_counts_cannot_substitute_for_replica_agreement(self):
        r=pilot.synthetic_replicas([.1]*8,[-.1]*8,[0.]*8,support=[1000000]*8)
        _,report=pilot.classify_tail(r,.75,5)
        self.assertNotEqual(report['tail_classification'],pilot.NONZERO)
