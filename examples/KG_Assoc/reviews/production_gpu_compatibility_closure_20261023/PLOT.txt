"""Plot actual exported compatibility summaries, without new estimators or synthetic observations."""
import argparse
import csv
from pathlib import Path
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

D = Path(__file__).resolve().parent
p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--input', type=Path, default=D / 'replay')
p.add_argument('--output', type=Path, required=True)
a = p.parse_args()
a.output.mkdir(exist_ok=False)
rows = list(csv.DictReader((a.input / 'PER_SYSTEM.csv').open()))
curves = list(csv.DictReader((a.input / 'STRESS_CHANNELS.csv').open()))
labels = [r['system'].replace('F04_N010_RP080_', '').replace('_', ' ') for r in rows]
fig, axes = plt.subplots(2, 2, figsize=(15, 10))
y = list(range(len(rows)))
axes[0, 0].barh(y, [float(r['G0']) for r in rows], color='#3274a1')
axes[0, 0].set_yticks(y, labels, fontsize=8)
axes[0, 0].invert_yaxis()
axes[0, 0].set_xlabel('Whole-test G(0), reduced units')
axes[0, 0].set_title('Finite stress output in every condition')
axes[0, 1].barh([v-.18 for v in y], [int(r['creations']) for r in rows], height=.35, label='Creations')
axes[0, 1].barh([v+.18 for v in y], [int(r['breaks']) for r in rows], height=.35, label='Breaks')
axes[0, 1].set_yticks(y, labels, fontsize=8)
axes[0, 1].invert_yaxis()
axes[0, 1].set_xscale('log')
axes[0, 1].set_xlabel('Accepted events in 100k steps (log scale)')
axes[0, 1].legend()
axes[0, 1].set_title('Exact event replay; sparse Ea10 activity remains sparse')
for ax, systems, title in [
    (axes[1, 0], [f'F04_N010_RP080_EE{ee:02d}_EA04' for ee in (4, 6, 8, 10, 12)], 'Ee series: saved whole-test modulus'),
    (axes[1, 1], [f'F04_N010_RP080_EE08_EA{ea:02d}' for ea in (4, 6, 8, 10)], 'Ea series: different qualified starts, one family each')]:
    for sid in systems:
        curve = [r for r in curves if r['system'] == sid]
        g0 = float(curve[0]['G'])
        label = sid.split('RP080_')[1].replace('_', ' ') + (' r003' if sid.endswith('EA10') else ' r001')
        ax.plot([float(r['time']) for r in curve], [float(r['G'])/g0 for r in curve], label=label, linewidth=1)
    ax.set_xscale('symlog', linthresh=.01)
    ax.set_yscale('symlog', linthresh=.01)
    ax.set_xlabel('Lag time, reduced units')
    ax.set_ylabel('G(t) / G(0), signed')
    ax.set_title(title)
    horizon = max(float(r['time']) for r in curve if int(r['n_pairs']) >= 8)
    ax.axvspan(horizon, float(curve[-1]['time']), color='grey', alpha=.15, label='<8 raw pairs')
    ax.legend(fontsize=8)
    ax.grid(alpha=.2)
fig.suptitle('Production compatibility only: one 100k test per condition; no terminal-tail or stationarity claim')
fig.tight_layout()
fig.savefig(a.output / 'COMPATIBILITY.png', dpi=160)
fig.savefig(a.output / 'COMPATIBILITY.pdf')
