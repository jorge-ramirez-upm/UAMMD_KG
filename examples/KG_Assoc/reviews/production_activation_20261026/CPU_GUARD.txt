"""Audit subprocess boundary: Git plus the exact authenticated CPU readback-dispatch probe."""
from pathlib import Path
import json
import hashlib
import subprocess

D = Path(__file__).resolve().parent


def guard():
    preserved = json.loads((D / 'PRESERVED.sha256').read_text())
    probes = {p: h for p, h in preserved.items()
              if Path(p).name == 'kg_assoc_stars_candidate_smoke'}
    assert probes, 'no authenticated CPU probe executable'
    original = subprocess.Popen
    def allowed(command, *args, **kwargs):
        if not isinstance(command, (list, tuple)) or not command:
            raise ValueError('CPU audit forbids shell commands')
        if command[0] != 'git':
            if not (len(command) == 2 and command[1] == '--readback-call-path-info'
                    and str(command[0]) in probes):
                raise ValueError('CPU audit forbids native/GPU/MD execution except the exact CPU probe')
            path = Path(command[0])
            if hashlib.sha256(path.read_bytes()).hexdigest() != probes[str(path)]:
                raise ValueError('CPU probe executable changed')
        return original(command, *args, **kwargs)
    subprocess.Popen = allowed
    return original
