    planned_steps=4500000000, prepared=states.count('prepared'),
    independently_validated=45-states.count('prepared'),
    preserved_bindings=len(F['c'].load_json(D / 'PRESERVED.sha256')),
    certifications=F['V']['CERTIFICATIONS'], unchanged_native_sources=True,
    protected_directories_unchanged=True, free_bytes=free,
    GPU_MD_steps_executed_by_assistant=0), indent=2, sort_keys=True))
